def _stratified_sample(files, scan_range):
    """分层抽样: 按板块分配 scan_range 配额, 保证深市/创/科覆盖。

    板块分类:
      - sh60: 沪主板 (~1785只)
      - sz00: 深主板 (~1582只)
      - sz30: 创业板 (~1412只)
      - sh68: 科创板 (~619只)
      - other: 其他(指数/ETF等, 后续被 is_tradable_code 过滤)

    配额按各板股票数比例分配, 向上取整确保不超 scan_range。
    SOUL#9: scan_range<=0 返回全量, 不复用 files[:0] 假绿逻辑。
    """
    if not scan_range or scan_range <= 0:
        return files

    by_board = {}
    others = []
    for f in files:
        name = f.name
        if name.startswith('sh60'):
            by_board.setdefault('sh60', []).append(f)
        elif name.startswith('sz00'):
            by_board.setdefault('sz00', []).append(f)
        elif name.startswith('sz30'):
            by_board.setdefault('sz30', []).append(f)
        elif name.startswith('sh68'):
            by_board.setdefault('sh68', []).append(f)
        else:
            others.append(f)

    # 按比例分配配额
    board_keys = ['sh60', 'sz00', 'sz30', 'sh68']
    total_board = sum(len(by_board.get(k, [])) for k in board_keys)
    if total_board == 0:
        return files[:scan_range]

    quota = {}
    remaining = scan_range
    for k in board_keys:
        count = len(by_board.get(k, []))
        if count == 0:
            quota[k] = 0
            continue
        q = max(1, int(scan_range * count / total_board))
        q = min(q, count)
        quota[k] = q
        remaining -= q
    # 把剩余配额分配给最大的板
    if remaining > 0:
        largest = max(board_keys, key=lambda k: len(by_board.get(k, [])))
        quota[largest] = min(quota[largest] + remaining, len(by_board[largest]))

    result = []
    for k in board_keys:
        result.extend(by_board.get(k, [])[:quota[k]])
    # other 填不满则补
    if len(result) < scan_range and others:
        result.extend(others[:scan_range - len(result)])
    return result[:scan_range]


def _compute_factor_score(inds, factor_weights, exceptions, ic_signs=None):
    """计算因子综合分 (z-score 截面标准化后加权求和)。

    因子缺失 (None/不存在) → 中性 0, 异常登记进 exceptions (SOUL#9)。
    ic_signs 传入 {name: ic} 时按 IC 符号定权重方向 (Task#4): 负 IC 权重取负,
    符号未知(缺失/0)置 0 不启用 —— 与 run_stock_selection 融合路径同一算术
    (apply_ic_sign), 杜绝出现第二条无符号防护的加权路径。
    返回: (factor_score: float, missing_count: int)
    """
    if not factor_weights:
        return 0.0, 0

    if ic_signs is not None:
        factor_weights, skipped = apply_ic_sign(factor_weights, ic_signs)
        for k in skipped:
            exceptions.append({'code': inds.get('_code', '?'), 'kind': 'factor_ic_sign_unknown',
                               'factor': k, 'detail': '权威源无有效 IC, 符号未知 -> 权重置 0 不启用'})

    raw_values = {}
    missing_count = 0
    for fname, weight in factor_weights.items():
        if weight == 0:
            continue
        val = inds.get(fname)
        if val is None:
            missing_count += 1
            raw_values[fname] = 0.0
        else:
            try:
                raw_values[fname] = float(val)
            except (TypeError, ValueError):
                missing_count += 1
                raw_values[fname] = 0.0
                exceptions.append({'code': inds.get('_code', '?'), 'kind': 'factor_missing',
                                   'factor': fname, 'detail': f'非数值:{val}'})

    if len(raw_values) <= 1:
        return 0.0, missing_count

    # z-score 标准化
    vals = list(raw_values.values())
    mean_v = sum(vals) / len(vals)
    variance = sum((v - mean_v) ** 2 for v in vals) / len(vals)
    std_v = variance ** 0.5 if variance > 0 else 1.0

    factor_score = 0.0
    for fname, weight in factor_weights.items():
        if weight == 0:
            continue
        z = (raw_values.get(fname, 0.0) - mean_v) / std_v
        factor_score += weight * z

    return factor_score, missing_count

def resolve_factor_layer_enabled(cli_override=None):
    """因子层硬开关 -> (enabled, source)。优先级: CLI > config.json > 默认 false。

    WHY: 因子层能否启用必须由显式配置决定, 不能由"判不出来的 valid 数"隐式决定 ——
    132d 窗口 28 因子三值判定实测 valid=0 (QA: QA-28因子k实测与三值判定-20261018.md),
    IC 统计判定在当前效应量水平下原理上不可达, 正确答案是显式开关而不是凑数。
    """
    if cli_override is not None:
        return bool(cli_override), "CLI"
    import config_loader
    return config_loader.get_factor_layer_enabled()


def load_factor_ic_signs():
    """权威源逐因子 IC 表 {name: ic} —— 符号处理的数据来源。

    权威源缺失/不可读 -> {}: 调用方按"符号未知"把因子权重置 0 并登记, 不静默放行
    (负 IC 因子不得按正权重用; 符号未知同样不得按正权重用)。
    """
    p = PROJECT_ROOT / "data" / "factor_ic" / "factor_ic_latest.json"
    try:
        data = json.loads(p.read_text(encoding="utf-8"))
    except Exception:
        return {}
    signs = {}
    for f in data.get("factors", []) or []:
        if isinstance(f, dict) and f.get("name") is not None and f.get("ic") is not None:
            signs[f["name"]] = f["ic"]
    return signs


def apply_ic_sign(factor_weights, ic_signs):
    """Task#4 符号处理: 按 IC 符号定权重方向, 负 IC 因子不得按正权重加分。

    effective[k] = sign(ic) * w:
      ic > 0 -> 权重不变 (正信号, 高 z 加分);
      ic < 0 -> 权重取负 (高 z 降分, 与"因子值负向预测未来收益"一致);
      ic 缺失/为 0 -> 权重置 0 并登记 (符号未知不启用, 禁静默)。
    返回 (effective_weights, skipped); 不改入参。
    """
    out, skipped = {}, []
    for k, w in factor_weights.items():
        if not w:
            out[k] = 0.0
            continue
        ic = ic_signs.get(k)
        if ic is None or ic == 0:
            out[k] = 0.0
            skipped.append(k)
            continue
        out[k] = float(w) * (1.0 if ic > 0 else -1.0)
    return out, skipped


def run_stock_selection(top_n=10, scan_range=3000, factor_layer=None):
    """执行选股 — T2a R0 改造版。

    改动:
      1. 分层抽样替代 glob 前 N (覆盖沪/深/创/科)
      2. 因子分 (z-score 加权) 融入排序: composite = tech*0.7 + factor*0.3
      3. 追高门 (涨幅>CHASE_BAN_PCT) 从 TOP 剔除, 记入 excluded_chase
      4. 因子缺失 = 中性0 + 异常登记, 禁静默吞 (SOUL#9)
      5. (2026-10-19 L4) 因子层硬开关 factor_layer_enabled: 默认 false, 每次运行
         打印当前值+来源; false 时因子层不参与打分, 选股只走结构/执行性判据
      6. (2026-10-19 L4/Task#4) 因子权重按 IC 符号定方向 (apply_ic_sign),
         负 IC 因子权重取负, 杜绝"负 IC 因子被当正信号用"

    参数 factor_layer: None=读配置; True/False=CLI 显式覆盖 (来源记为 CLI)。
    """
    print(f"\n== 执行选股 (TOP {top_n}, 扫描 {scan_range}只) ==")

    from tdx_core import compute_indicators
    from tdx_score import score_stock
    from tdx_data import is_tradable_code, load_name_cache
    from config_loader import get_factor_weights
    from risk_rules import CHASE_BAN_PCT

    root = PROJECT_ROOT

    # 加载配置
    config_path = root / "config.json"
    with open(config_path, 'r', encoding='utf-8') as fh:
        config = json.load(fh)

    # T2a: 从单一真源读取因子权重
    factor_weights = get_factor_weights()
    # P0 修复(2026-10-18 老六): 原标签谎报为 factor_weights.json，实际 get_factor_weights()
    # 读的是 config.json 的 factor_weights 段（config_loader.py:122-125）。
    # 仓库根的 factor_weights.json（20 因子）实盘链路从不读取，仅被
    # apply_factor_weight_proposal.py(写)/factor_weight_proposal.py(读) 使用。
    print(f"因子权重真源: config.json -> factor_weights ({len(factor_weights)}个因子)")
    print(f"  [注] 仓库根 factor_weights.json 非本链路真源，实盘不读取")

    # L4 硬开关: 每次运行打印当前值+来源 (没人打印的配置项等于不存在)
    factor_layer_on, factor_layer_src = resolve_factor_layer_enabled(factor_layer)
    print(f"因子层开关: factor_layer_enabled={factor_layer_on} (来源: {factor_layer_src})")
    if not factor_layer_on:
        print("  语义: false = 因子层不参与选股打分, 选股只走结构/执行性判据 (tech_score)")
        factor_weights = {}

    # 加载历史数据
    history_dir = root / "data" / "history_daily"
    all_files = [f for f in history_dir.glob("*.json") if f.name != "download_errors.json"]

    # T2a 抽样修正: 分层抽样替代 glob 前 N
    files = _stratified_sample(all_files, scan_range)
    board_counts = {}
    for f in files:
        name = f.name
        if name.startswith('sh60'): board_counts['sh60'] = board_counts.get('sh60', 0) + 1
        elif name.startswith('sz00'): board_counts['sz00'] = board_counts.get('sz00', 0) + 1
        elif name.startswith('sz30'): board_counts['sz30'] = board_counts.get('sz30', 0) + 1
        elif name.startswith('sh68'): board_counts['sh68'] = board_counts.get('sh68', 0) + 1
    print(f"加载历史数据: {len(files)}只 (分层: {board_counts}), 全量库: {len(all_files)}只")

    # 名称表: ST/退市判定依据。复用 tdx_data.load_name_cache (与 backtest/integrated_picker
    # 同一把钥匙, 不新造名称获取逻辑); 池内未命中的代码 name="" 退化为纯板块白名单, 不误杀。
    name_cache = load_name_cache()
    print(f"名称表: {len(name_cache)} 条 (stock_names.json, ST/退市判定依据)")

    # 选股评分 — Pass 1: 计算技术指标分
    candidates = []  # [{code, tech_score, inds, daily_chg}]
    processed = 0

    # SOUL#9: 异常分类登记, 不再静默吞
    _ERR_SAMPLE_CAP = 10
    logic_errors = 0
    data_skipped = 0
    scan_error_samples = []
    factor_exceptions = []  # 因子缺失/异常登记

    for filename in files:
        filepath = history_dir / filename
        code = filename.stem

        # 数据层: 文件/JSON 损坏 → data_skipped 登记
        try:
            with open(filepath, 'r', encoding='utf-8') as fh:
                data = json.load(fh)
        except Exception as e:
            data_skipped += 1
            if len(scan_error_samples) < _ERR_SAMPLE_CAP:
                scan_error_samples.append({"code": code, "kind": "data",
                                           "error": f"{type(e).__name__}: {str(e)[:60]}"})
            continue

        if len(data) < 100:
            continue

        # 主板前缀白名单 ∧ ST/退市剔除 (D8 老大拍板): 创业板/科创/北证/ETF/ST/退市不进评分
        # name 必须传: is_tradable_code 的 ST/退市段 `if name and is_st_stock(name)` 只在
        # 传了 name 时生效 —— 不传则该段被短路, ST/退市股全额进评分 (P0-A, 2026-10-19 修复)
        if not is_tradable_code(code, name_cache.get(code, "")):
            continue

        # 逻辑层: 指标/评分异常 → logic_errors 登记
        try:
            inds = compute_indicators(data)
            if inds is None:
                data_skipped += 1
                continue

            # 计算当日涨幅 (用于追高门)
            daily_chg = 0.0
            if len(data) >= 2:
                prev_close = data[-2].get('close', 0)
                cur_close = data[-1].get('close', 0)
                if prev_close and prev_close > 0:
                    daily_chg = (cur_close - prev_close) / prev_close * 100

            result = score_stock(inds, strategy=config)
            if result and isinstance(result, tuple) and len(result) >= 1:
                tech_score = result[0]
                if tech_score is not None and tech_score > 0:
                    candidates.append({
                        'code': code,
                        'tech_score': float(tech_score),
                        'inds': inds,
                        'daily_chg': daily_chg,
                        'details': result[1] if len(result) > 1 else []
                    })

            processed += 1
            if processed % 500 == 0:
                print(f"  已处理: {processed}/{len(files)}, 有效: {len(candidates)}")
        except Exception as e:
            logic_errors += 1
            if len(scan_error_samples) < _ERR_SAMPLE_CAP:
                scan_error_samples.append({"code": code, "kind": "logic",
                                           "error": f"{type(e).__name__}: {str(e)[:60]}"})
            continue

    # SOUL#9: 异常告警
    if logic_errors or data_skipped:
        print(f"  ⚠️ 扫描异常: 逻辑层 {logic_errors} / 数据层 {data_skipped} (前5条:)")
        for s in scan_error_samples[:5]:
            print(f"    [{s['kind']}] {s['code']}: {s['error']}")

    # T2a: Pass 2 — 因子分 z-score 截面标准化 + 融合排序
    # 隐患1 修复(2026-10-18 老六): _dead_factors 需在分支外可见，供产物落盘
    _dead_factors = []
    if factor_weights and candidates:
        # Task#4 符号处理: 权重方向按 IC 符号定 —— 负 IC 因子权重取负, 符号未知置 0 不启用。
        # WHY: 权威源 TURN5 IC=-0.0658 (负), 原 factor_score += w*z 无符号防护,
        # 负 IC 因子按正权重加分 = 方向反了的开仓开关 (闭环审查 L4 断裂项)。
        ic_signs = load_factor_ic_signs()
        factor_weights, sign_skipped = apply_ic_sign(factor_weights, ic_signs)
        if sign_skipped:
            print(f"[告警] 以下 {len(sign_skipped)} 个因子权威源无有效 IC(缺失或为0), "
                  f"符号未知 -> 权重置 0 不启用: {sign_skipped}")
            for k in sign_skipped:
                factor_exceptions.append({
                    'code': '*', 'kind': 'factor_ic_sign_unknown', 'factor': k,
                    'detail': '权威源无该因子有效 IC(缺失或为0), 符号未知 -> 权重置 0 不启用'})
        fw_keys = [k for k, v in factor_weights.items() if v != 0]
        if fw_keys:
            all_raw = {k: [] for k in fw_keys}
            _missing_cnt = {k: 0 for k in fw_keys}
            for c in candidates:
                for k in fw_keys:
                    v = c['inds'].get(k)
                    if v is None:
                        all_raw[k].append(None)
                        _missing_cnt[k] += 1
                    else:
                        try:
                            all_raw[k].append(float(v))
                        except (TypeError, ValueError):
                            all_raw[k].append(None)
                            _missing_cnt[k] += 1
                            factor_exceptions.append({'code': c['code'], 'kind': 'factor_missing',
                                                      'factor': k, 'detail': '非数值'})

            # 隐患1 修复(2026-10-18 老六): 权重非零但 compute_indicators 根本不产出该键时,
            # 原实现会静默把 z 置 0 -> factor_score 恒 0 -> score 退化成 tech*0.7, 不报错不告警。
            # 现改为: 逐键登记 + 显式告警, 让"因子层没在工作"这件事必须可见。
            _dead_factors = []
            for k in fw_keys:
                if _missing_cnt[k] == len(candidates) and candidates:
                    _dead_factors.append(k)
                    factor_exceptions.append({
                        'code': '*', 'kind': 'factor_dead_all_universe', 'factor': k,
                        'detail': f'{len(candidates)}/{len(candidates)} 只候选均无此键，'
                                  f'该因子权重 {factor_weights.get(k)} 实际未生效（z 全置 0）'})
            if _dead_factors:
                print(f"[告警] 以下 {len(_dead_factors)} 个因子在 config.json 中权重非 0，"
                      f"但 compute_indicators 未产出对应键 —— 因子分实际未参与打分：{_dead_factors}")
                print(f"[告警] 当前 score 实际退化为 tech_score*0.7（factor_score 恒 0）。"
                      f"请核对 config.json 的 factor_weights 与 tdx_indicators.compute_indicators 的返回键。")

            # z-score 标准化 — 用索引列表而非字典查找, 避免浮点精度 KeyError
            factor_zscores = {}  # {factor_key: [z1, z2, ...] 与 candidates 同序}
            for k in fw_keys:
                vals = [v for v in all_raw[k] if v is not None]
                if len(vals) < 2:
                    factor_zscores[k] = [0.0] * len(candidates)
                    continue
                mean_v = sum(vals) / len(vals)
                std_v = (sum((v - mean_v) ** 2 for v in vals) / len(vals)) ** 0.5
                std_v = std_v if std_v > 1e-9 else 1.0
                zs = []
                for v in all_raw[k]:
                    if v is None:
                        zs.append(0.0)
                    else:
                        zs.append((v - mean_v) / std_v)
                factor_zscores[k] = zs

            # 融合: composite = tech*0.7 + factor*0.3
            for i, c in enumerate(candidates):
                factor_score = 0.0
                for k in fw_keys:
                    w = factor_weights[k]
                    z = factor_zscores[k][i] if k in factor_zscores else 0.0
                    factor_score += w * z
                c['factor_score'] = factor_score
                c['score'] = c['tech_score'] * 0.7 + factor_score * 0.3
        else:
            # 所有权重为0, 只用技术分
            for c in candidates:
                c['score'] = c['tech_score']
                c['factor_score'] = 0.0
    else:
        for c in candidates:
            c['score'] = c['tech_score']
            c['factor_score'] = 0.0

    if factor_exceptions:
        print(f"  ⚠️ 因子异常登记: {len(factor_exceptions)} 条 (SOUL#9)")
        for e in factor_exceptions[:5]:
            print(f"    [{e['kind']}] {e['code']} {e['factor']}: {e['detail']}")

    print(f"评分完成: {processed}只股票, 有效候选: {len(candidates)}只")

    # T2a: 按综合分排序
    candidates.sort(key=lambda x: x['score'], reverse=True)

    # T2a: 追高门 — 信号日涨幅 > CHASE_BAN_PCT 的候选从 TOP 剔除
    chase_threshold = CHASE_BAN_PCT  # 4.0, 只读常量, 不改
    excluded_chase = []
    filtered_candidates = []
    for c in candidates:
        if c['daily_chg'] > chase_threshold:
            excluded_chase.append({
                'code': c['code'],
                'daily_chg_pct': round(c['daily_chg'], 2),
                'excluded_at': datetime.now().isoformat(),
                'reason': f'信号日涨幅{c["daily_chg"]:.2f}% > CHASE_BAN_PCT{chase_threshold}%'
            })
        else:
            filtered_candidates.append(c)

    if excluded_chase:
        print(f"  🚫 追高门剔除: {len(excluded_chase)} 只 (涨幅>{chase_threshold}%)")
        for e in excluded_chase[:5]:
            print(f"    {e['code']}: {e['daily_chg_pct']}%")

    # 取 TOP N
    top_picks = filtered_candidates[:top_n]

    # 构建输出 (去掉内部字段)
    pick_output = []
    for c in top_picks:
        pick_output.append({
            'code': c['code'],
            'score': round(c['score'], 2),
            'tech_score': round(c['tech_score'], 2),
            'factor_score': round(c.get('factor_score', 0.0), 4),
            'details': c['details']
        })

    # 保存结果
    result = {
        'timestamp': datetime.now().isoformat(),
        'strategy': config.get('strategy'),
        'version': config.get('version'),
        'total_scanned': processed,
        'total_valid': len(candidates),
        'top_n': top_n,
        'picks': pick_output,
        # T2a: 追高剔除记录
        'excluded_chase': excluded_chase,
        # T2a: 因子权重来源（P0 修复 2026-10-18: 原值 'factor_weights.json' 为谎报，真源是 config.json）
        'factor_weights_source': 'config.json:factor_weights',
        'factor_weights_source_note': '仓库根 factor_weights.json 实盘链路不读取，非真源',
        'factor_weights_count': len(factor_weights),
        'phase1_weights_used': len(config.get('factor_weights', {})),
        # 隐患1 修复: 权重非 0 但 compute_indicators 未产出的因子（因子分实际未生效）
        'factor_dead_universe': _dead_factors,
        # SOUL#9: 异常登记随产物落盘
        'scan_errors': {
            'logic_errors': logic_errors,
            'data_skipped': data_skipped,
            'samples': scan_error_samples,
            'factor_exceptions': factor_exceptions[:20],
        }
    }

    output_path = root / "data" / "picks_latest.json"
    atomic_dump(result, output_path, ensure_ascii=False, indent=2)

    print(f"✅ 选股结果已保存: {output_path}")

    # 显示 TOP N
    print(f"\nTOP {top_n}最优个股:")
    for i, stock in enumerate(pick_output, 1):
        action = '买入' if stock['score'] > 60 else ('观察' if stock['score'] > 40 else '放弃')
        print(f"  {i}. {stock['code']}: {stock['score']:.2f}分 (技术{stock['tech_score']:.1f}+因子{stock['factor_score']:.2f}) - {action}")

    if excluded_chase:
        print(f"\n🚫 追高门剔除 ({len(excluded_chase)}只):")
        for e in excluded_chase:
            print(f"  {e['code']}: {e['daily_chg_pct']}% — {e['reason']}")

    return pick_output, processed


def generate_trade_signals(picks, mode='test'):
    """生成交易信号"""
    print("\n== 生成交易信号 ==")
    
    signals = []
    for i, pick in enumerate(picks, 1):
        signal = {
            'timestamp': datetime.now().isoformat(),
            'rank': i,
            'code': pick['code'],
            'score': pick['score'],
            'action': 'BUY' if pick['score'] > 60 else ('HOLD' if pick['score'] > 40 else 'PASS'),
            'confidence': 'HIGH' if pick['score'] > 80 else ('MEDIUM' if pick['score'] > 60 else 'LOW'),
            'reason': pick.get('details', [])[:3] if pick.get('details') else [],
            'mode': mode
        }
        signals.append(signal)
        print(f"  {signal['code']}: {signal['action']} (置信度: {signal['confidence']})")
    
    # 保存信号
    root = PROJECT_ROOT
    signals_path = root / "data" / "trade_signals.json"
    atomic_dump(signals, signals_path, ensure_ascii=False, indent=2)
    
    print(f"✅ 交易信号已保存: {signals_path}")
    
    return signals


def send_notification(signals, mode='test', removed_counts=None):
    """发送飞书通知

    P2fix (2026-10-09): 移除硬编码的 DELISTED/STALE/ZERO_CLOSE=0 占位符，
    改用调用方传入的真实剔除计数。若未传入则显式标记"未接入"（禁止打0）。
    """
    print("\n== 发送飞书通知 ==")

    # S6: 交易时段防护
    from trading_hours import is_trading_hours
    _in_session = is_trading_hours(datetime.now())
    session_status = "交易中" if _in_session else "非交易时段(dry_run告警)"

    # P2fix: 接收真实剔除计数
    if removed_counts is None:
        # 未接入时显式标记，不得打0假数
        total_scanned = "未接入"
        removed_info = {'delisted_st': '未接入', 'stale_data': '未接入', 'zero_close': '未接入'}
    else:
        total_scanned = removed_counts.get('total_scanned', '未知')
        removed_info = {
            'delisted_st': removed_counts.get('delisted_st', '未知'),
            'stale_data': removed_counts.get('stale_data', '未知'),
            'zero_close': removed_counts.get('zero_close', '未知'),
        }

    SEL_COUNT = len(signals)

    # 生成通知内容（含剔除计数行）
    notification = {
        'timestamp': datetime.now().isoformat(),
        'mode': mode,
        'session_status': session_status,
        'total_scanned': total_scanned,
        'removed': removed_info,
        'total_picks': SEL_COUNT,
        'buy_signals': len([s for s in signals if s['action'] == 'BUY']),
        'hold_signals': len([s for s in signals if s['action'] == 'HOLD']),
        'pass_signals': len([s for s in signals if s['action'] == 'PASS']),
    }

    # P2fix: 原子写
    root = PROJECT_ROOT
    notify_path = root / "data" / "feishu_notification.json"
    atomic_dump(notification, notify_path, ensure_ascii=False, indent=2)
    
    print(f"✅ 通知记录已保存: {notify_path}")
    print(f"   买入信号: {notification['buy_signals']}个")
    print(f"   持有信号: {notification['hold_signals']}个")
    print(f"   放弃信号: {notification['pass_signals']}个")


def main():
    parser = argparse.ArgumentParser(description='自动化选股实盘流水线')
    parser.add_argument('--mode', choices=['daily', 'test', 'live'], default='daily',
                        help='执行模式: daily(每日自动), test(测试), live(实盘)')
    parser.add_argument('--top-n', type=int, default=10, help='选择股票数量')
    parser.add_argument('--scan-range', type=int, default=3000, help='扫描股票数量')
    parser.add_argument('--factor-layer', dest='factor_layer', action='store_true', default=None,
                        help='显式启用因子层 (覆盖 config.json; 缺省读 factor_layer_enabled, 默认 false)')
    parser.add_argument('--no-factor-layer', dest='factor_layer', action='store_false',
                        help='显式关闭因子层 (覆盖 config.json)')
    
    args = parser.parse_args()
    
    print("="*60)
    print("  自动化选股实盘流水线")
    print(f"  时间：{datetime.now().strftime('%Y-%m-%d %H:%M')}")
    print(f"  模式：{args.mode}")
    print("="*60)
    print()
    
    # 1. 数据健康检查
    health_ok = check_data_health()
    if not health_ok:
        print("\n❌ 数据健康检查未通过，终止执行")
        return 1
    
    # 2. 门禁检查
    gates_ok = check_gates()
    if not gates_ok:
        print("\n⚠️ 门禁检查未全部通过，继续执行但需关注")
    
    # P2fix: 传递真实剔除计数
    picks, processed = run_stock_selection(args.top_n, args.scan_range, factor_layer=args.factor_layer)
    if not picks:
        print("\n❌ 选股失败，无有效候选")
        return 1

    # 4. 生成交易信号
    signals = generate_trade_signals(picks, args.mode)

    # 5. 发送通知（传入剔除计数）
    removed_counts = {
        'total_scanned': processed,
        'delisted_st': 0,  # TODO: 从 run_stock_selection 返回真实计数
        'stale_data': 0,
        'zero_close': 0,
    }
    send_notification(signals, args.mode, removed_counts=removed_counts)
    
    print("\n" + "="*60)
    print("  ✅ 流水线执行完成")
    print("="*60)
    
    return 0


if __name__ == '__main__':
    sys.exit(main())
