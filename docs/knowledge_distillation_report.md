# 分歧买点：烂板弱转强
def identify_divergence_buy(signal):
    conditions = [
        signal.pre_day_limit_up_breaks > 3,  # 前日炸板>3次
        signal.today_open > signal.yesterday_close * 1.03,  # 高开>3%
        signal.volume_ratio > 0.05,  # 竞价成交量>前日5%
        signal.break_time < "09:45",  # 15分钟内翻红
    ]
    return all(conditions)

# 一致卖点：缩量加速→放量滞涨
def identify_consensus_sell(signal):
    conditions = [
        signal.turnover_rate < 0.10,  # 换手率<10%（缩量加速到极致）
        signal.volume_spike > 1.5,     # 成交量放大50%
        signal.price_change < 0.02,    # 涨幅≤2%（放量滞涨）
    ]
    return all(conditions)

```python
def position_sizing(regime, portfolio_value):
    """根据市场状态调整仓位"""
    regime_rules = {
        '上涨趋势': 0.8,
        '下跌趋势': 0.1,
        '宽幅震荡': 0.3,
        '窄幅震荡': 0.5,
    }
    base_ratio = regime_rules.get(regime, 0.3)
    return min(portfolio_value * base_ratio, portfolio_value)
```

## 四、风险提示

1. **因子衰减**：历史有效因子可能因市场结构变化失效，需持续IC监控
2. **过拟合风险**：深度学习模型在样本外可能失效，需正交风险因子抑制
3. **极端行情**：黑天鹅事件下所有规则可能失效，保留人工干预通道
4. **执行延迟**：龙头战法需要快速执行，系统响应时间<100ms
